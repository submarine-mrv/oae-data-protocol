    name: PlatformType
description: Platform categories from the SeaVoX L06 vocabulary.
from_schema: https://schema.oaedata.org/OAEDataSchema
rank: 1000
permissible_values:
  aeroplane:
    text: aeroplane
    description: A fixed-wing self-propelled aircraft.
    meaning: NVS_L06:62/
    title: aeroplane
  autonomous_surface_water_vehicle:
    text: autonomous_surface_water_vehicle
    description: A programmed or remotely controlled self-propelled vessel operating
      on the surface of water bodies with no human occupants; includes saildrones.
    meaning: NVS_L06:3B/
    title: autonomous surface water vehicle
  autonomous_underwater_vehicle:
    text: autonomous_underwater_vehicle
    description: A free-roving platform operating in the water column with propulsion
      but no human operator on board (e.g. Autosub, Gavia).
    meaning: NVS_L06:25/
    title: autonomous underwater vehicle
  beach_intertidal_zone_structure:
    text: beach_intertidal_zone_structure
    description: A structure to which instrumentation may be attached that is either
      in air or under water depending on the state of the tide and weather conditions.
    meaning: NVS_L06:13/
    title: beach/intertidal zone structure
  coastal_structure:
    text: coastal_structure
    description: A fixed man-made structure permanently linked to land with access
      to water at all states of the tide to which instrumentation may be attached
      (e.g. pier).
    meaning: NVS_L06:17/
    title: coastal structure
  diver:
    text: diver
    description: A human being with self-contained equipment or surface-connected
      suit enabling operation within the water column.
    meaning: NVS_L06:72/
    title: diver
  drifting_subsurface_float:
    text: drifting_subsurface_float
    description: An unmanned instrumented platform drifting freely in the water column
      at a depth governed by its density (e.g. Swallow float).
    meaning: NVS_L06:44/
    title: drifting subsurface float
  drifting_subsurface_profiling_float:
    text: drifting_subsurface_profiling_float
    description: An unmanned instrumented platform drifting freely in the water column
      that periodically makes vertical traverses through the water column (e.g. Argo
      float).
    meaning: NVS_L06:46/
    title: drifting subsurface profiling float
  drifting_surface_float:
    text: drifting_surface_float
    description: An unmanned instrumented platform operating on the surface of the
      water column often attached to a drogue to track currents rather than winds
      (e.g. Argos buoy).
    meaning: NVS_L06:42/
    title: drifting surface float
  fishing_vessel:
    text: fishing_vessel
    description: A platform operating on the surface of the water column whose primary
      purpose is the commercial harvesting of fish or shellfish but may be engaged
      in scientific activities such as fish stock surveys or mooring deployments and
      recoveries.
    meaning: NVS_L06:36/
    title: fishing vessel
  fixed_benthic_node:
    text: fixed_benthic_node
    description: A collection of oceanographic instruments mounted at a fixed position
      on the seabed (e.g. POL Monitoring Platform, seabed ADCP).
    meaning: NVS_L06:11/
    title: fixed benthic node
  fixed_subsurface_vertical_profiler:
    text: fixed_subsurface_vertical_profiler
    description: A platform that periodically makes an automated vertical traverse
      of the water column at a predetermined fixed location. (e.g. YSI vertical profiler,
      HOMER CTD).
    meaning: NVS_L06:45/
    title: fixed subsurface vertical profiler
  float:
    text: float
    description: A free-floating platform either on the surface of the water column
      or at a predetermined depth within the water column.
    meaning: NVS_L06:47/
    title: float
  glider:
    text: glider
    description: A fixed-wing aircraft with no propulsion.
    meaning: NVS_L06:6A/
    title: glider
  helicopter:
    text: helicopter
    description: An aircraft without wings that obtains its lift from the rotation
      of overhead blades.
    meaning: NVS_L06:67/
    title: helicopter
  human:
    text: human
    description: A human being without specialised equipment operating on land or
      the surface of the water column.
    meaning: NVS_L06:71/
    title: human
  land_or_seafloor:
    text: land_or_seafloor
    description: A platform located on the solid surface of the Earth either above
      or below sea level.
    meaning: NVS_L06:10/
    title: land or seafloor
  land_onshore_structure:
    text: land_onshore_structure
    description: A fixed man-made structure on land to which instrumentation may be
      attached (e.g. meteorological tower).
    meaning: NVS_L06:14/
    title: land/onshore structure
  land_onshore_vehicle:
    text: land_onshore_vehicle
    description: An instrumented vehicle or sample collector that operates on the
      solid surface of the Earth (e.g. mobile meteorological station, land crawler,
      snowmobile).
    meaning: NVS_L06:15/
    title: land/onshore vehicle
  lowered_unmanned_submersible:
    text: lowered_unmanned_submersible
    description: An unmanned platform lowered and raised vertically by a cable from
      the mothership.  Includes any type of profiling sensor mounting such as CTD
      frames, profiling radiometers and instrumented nets.
    meaning: NVS_L06:26/
    title: lowered unmanned submersible
  man_powered_boat:
    text: man_powered_boat
    description: A platform operating on the surface of the water column that is manually
      propelled and may not be easily removed from the water (e.g. trireme).
    meaning: NVS_L06:38/
    title: man-powered boat
  man_powered_small_boat:
    text: man_powered_small_boat
    description: A platform operating on the surface of the water column that is manually
      propelled and may be easily removed from the water (e.g. rowing boat, canoe).
    meaning: NVS_L06:3A/
    title: man-powered small boat
  mesocosm_bag:
    text: mesocosm_bag
    description: A large polythene bag containing a water sample suspended in the
      natural environment so that it shares ambient physical conditions such as temperature
      and light levels, but is chemically isolated to allow experiments such as fertilisation.
    meaning: NVS_L06:19/
    title: mesocosm bag
  mesocosm_tank:
    text: mesocosm_tank
    description: An indoor or outdoor tank containing water, used to house an experimental
      system. Instruments attached to the tank are used to measure and control conditions
      to help evaluate how organisms or communities react to controlled environmental
      changes.
    meaning: NVS_L06:99/
    title: mesocosm tank
  moored_surface_buoy:
    text: moored_surface_buoy
    description: An unmanned instrumented platform operating on the surface of the
      water column loosely tethered to the seafloor to maintain a fixed position (e.g.
      ODAS buoy).
    meaning: NVS_L06:41/
    title: moored surface buoy
  mooring:
    text: mooring
    description: A tethered collection of oceanographic instruments at a fixed location
      that may include seafloor, mid-water and surface components.
    meaning: NVS_L06:48/
    title: mooring
  offshore_structure:
    text: offshore_structure
    description: A fixed (for the duration of the measurements) man-made structure
      away from the coast to which instrumentation may be attached (e.g. oil rig,
      gas rig or jack-up barge).
    meaning: NVS_L06:16/
    title: offshore structure
  propelled_unmanned_submersible:
    text: propelled_unmanned_submersible
    description: A platform operating in the water column attached to a mothership
      by an umbilical with limited propulsion and no human operator on board (e.g.
      ROV).
    meaning: NVS_L06:22/
    title: propelled unmanned submersible
  research_aeroplane:
    text: research_aeroplane
    description: A fixed-wing self-propelled aircraft that is equipped, manned and
      operated for atmospheric, meteorological or oceanographic research, sometimes
      also called 'instrumented platform'.
    meaning: NVS_L06:61/
    title: research aeroplane
  research_vessel:
    text: research_vessel
    description: A platform of any size operating on the surface of the water column
      in unpredictable locations that is specifically equipped, manned and operated
      for scientific, usually oceanographic, research.
    meaning: NVS_L06:31/
    title: research vessel
  river_station:
    text: river_station
    description: An instrumented structure in a river upstream of its tidal limit.
    meaning: NVS_L06:18/
    title: river station
  satellite:
    text: satellite
    description: A vehicle operating beyond the Earth's atmosphere without human occupants
      that orbits the Earth.
    meaning: NVS_L06:68/
    title: satellite
  sea_bed_vehicle:
    text: sea_bed_vehicle
    description: An instrumented platform that is propelled on wheels or tracks on
      the seabed (e.g benthic crawler).
    meaning: NVS_L06:12/
    title: sea bed vehicle
  self_propelled_boat:
    text: self_propelled_boat
    description: A small self-propelled platform operating on the surface of the water
      column in unpredictable locations that is smaller than a ship, but too large
      to easily remove from the water.
    meaning: NVS_L06:37/
    title: self-propelled boat
  self_propelled_small_boat:
    text: self_propelled_small_boat
    description: A small self-propelled platform operating on the surface of the water
      column that may be easily removed from the water (e.g. shore-based RIBs, ships'
      boats).
    meaning: NVS_L06:33/
    title: self-propelled small boat
  ship:
    text: ship
    description: 'A large platform operating on the surface of the water column. Objective
      definitions for guidelines: >50m length (EU), >100 foot length (USA), >300 GRT
      weight (SOLAS). Subjective definition: a ship is a vessel big enough to carry
      a boat.'
    meaning: NVS_L06:30/
    title: ship
  sub_surface_gliders:
    text: sub_surface_gliders
    description: Platforms with buoyancy-based propulsion that are capable of operations
      at variable depths which are not constrained to be near the sea surface.
    meaning: NVS_L06:27/
    title: sub-surface gliders
  submersible:
    text: submersible
    description: A platform operating within a water body.
    meaning: NVS_L06:20/
    title: submersible
  subsurface_mooring:
    text: subsurface_mooring
    description: A collection of oceanographic instruments attached to wires suspended
      between anchors on the seabed and buoyant spheres in the water column.
    meaning: NVS_L06:43/
    title: subsurface mooring
  surface_gliders:
    text: surface_gliders
    description: Platforms operating at a single depth near the sea surface, using
      a combination of solar energy and wave motion as means of propulsion.
    meaning: NVS_L06:3C/
    title: surface gliders
  surface_vessel:
    text: surface_vessel
    description: A mobile platform with propulsion operating on and restricted to
      the surface of a water body.
    meaning: NVS_L06:3Z/
    title: surface vessel
  towed_unmanned_submersible:
    text: towed_unmanned_submersible
    description: A vehicle towed by rigid cable through the water column at fixed
      or varying depth with no propulsion and no human operator (e.g. Towfish, Scanfish,
      UOR, SeaSoar).
    meaning: NVS_L06:23/
    title: towed unmanned submersible
  unmanned_aerial_vehicle:
    text: unmanned_aerial_vehicle
    description: 'Any untethered heavier-than-air aircraft that is not occupied by
      people: may be a remotely piloted aircraft or an autonomous aircraft. Also referred
      to as a drone.'
    meaning: NVS_L06:6D/
    title: unmanned aerial vehicle
  vessel_at_fixed_position:
    text: vessel_at_fixed_position
    description: A platform of any size occupying a fixed location on the surface
      of the water column for prolonged periods collecting scientific (oceanographic
      and meteorological) data as a primary or secondary mission.  Includes light
      vessels and weather ships.
    meaning: NVS_L06:34/
    title: vessel at fixed position
